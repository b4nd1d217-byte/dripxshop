  async function dripxZipOpenDiagnostic(file,error){
    const original=String(error?.message||error||'Error desconocido');
    let hasEocd=false,hasZipSignature=false;
    try{
      const head=new Uint8Array(await file.slice(0,Math.min(file.size,4096)).arrayBuffer());
      for(let i=0;i+3<head.length;i++){
        if(head[i]===0x50&&head[i+1]===0x4b&&(
          (head[i+2]===0x03&&head[i+3]===0x04)||
          (head[i+2]===0x05&&head[i+3]===0x06)||
          (head[i+2]===0x06&&head[i+3]===0x06)||
          (head[i+2]===0x07&&head[i+3]===0x08)
        )){hasZipSignature=true;break;}
      }
      const tailStart=Math.max(0,file.size-65557);
      const tail=new Uint8Array(await file.slice(tailStart).arrayBuffer());
      for(let i=Math.max(0,tail.length-65557);i+3<tail.length;i++){
        if(tail[i]===0x50&&tail[i+1]===0x4b&&tail[i+2]===0x05&&tail[i+3]===0x06){
          hasEocd=true;break;
        }
      }
    }catch{}
    if(!hasZipSignature&&!hasEocd){
      return new Error('El archivo seleccionado no contiene una firma ZIP reconocible. Comprueba que elegiste el .zip real y no un .rar, .7z, .apk renombrado o una descarga HTML.');
    }
    if(hasZipSignature&&!hasEocd){
      return new Error('El archivo empieza como ZIP, pero no contiene el registro final del directorio central (EOCD). Está incompleto o se descargó/compartió antes de terminar de comprimirse. Vuelve a crear o descargar el ZIP completo y revisa su tamaño. No es un error de meta.xml, Lua ni de la licencia.');
    }
    return new Error('Se encontró el cierre ZIP, pero no se pudo interpretar su directorio central. Puede estar dañado, protegido con cifrado no compatible o usar un formato dividido. Detalle del lector: '+original);
  }
  async function dripxZipReadEntry(entry,type,options={}){
    if(entry.encrypted)throw new Error('El archivo '+entry.filename+' está cifrado. Vuelve a crear el ZIP sin contraseña para que DRIP X SHOP pueda protegerlo.');
    const writer=new window.zip.Uint8ArrayWriter();
    const bytes=await entry.getData(writer,{
      checkCrc32:true,
      checkOverlappingEntry:true,
      strictness:'balanced',
      useWebWorkers:false,
      onprogress:(current,total)=>options.onProgress?.(current,total)
    });
    if(type==='string')return new TextDecoder('utf-8',{fatal:false}).decode(bytes);
    if(type==='uint8array')return bytes;
    throw new Error('Tipo de lectura ZIP no compatible: '+String(type));
  }
  function dripxZipWrapEntry(entry){
    const name=String(entry.filename||'');
    return {
      name,
      dir:!!entry.directory,
      unsafeOriginalName:name,
      unixPermissions:entry.unixMode,
      dosPermissions:entry.msdosAttributesRaw,
      date:entry.lastModDate,
      encrypted:!!entry.encrypted,
      _zipEntry:entry,
      _data:{uncompressedSize:Number(entry.uncompressedSize)||0,length:Number(entry.uncompressedSize)||0,crc32:entry.crc32},
      async(type,options){return dripxZipReadEntry(entry,type,options||{});}
    };
  }
  async function dripxZipReadDirectory(file){
    if(!window.zip?.ZipReader||!window.zip?.BlobReader||!window.zip?.Uint8ArrayWriter){
      throw new Error('No se cargó el nuevo lector ZIP. Cierra y vuelve a abrir Productos para actualizar sus dependencias.');
    }
    const reader=new window.zip.ZipReader(new window.zip.BlobReader(file),{
      checkOverlappingEntry:true,
      strictness:'balanced'
    });
    try{
      const entries=await reader.getEntries({strictness:'balanced'});
      await reader.close();
      return entries.map(dripxZipWrapEntry);
    }catch(error){
      try{await reader.close();}catch{}
      throw await dripxZipOpenDiagnostic(file,error);
    }
  }
  function dripxZipMtaBytecode(bytes){return !!bytes&&bytes.length>=4&&bytes[0]===0x1c&&bytes[1]===0x4d&&bytes[2]===0x76&&bytes[3]===0x62;}

    if(!file||!Number.isFinite(Number(file.size))||Number(file.size)<0)throw new Error('Selecciona un archivo ZIP válido.');
    if(!/\.zip$/i.test(String(file.name||'')))throw new Error('El archivo debe tener extensión .zip.');

    if(allEntries.length>LIMITS.entries)throw new Error('El ZIP contiene demasiadas entradas, incluidas carpetas.');
    if(allEntries.filter(entry=>!entry.dir).length>LIMITS.files)throw new Error('El ZIP supera el límite de 3.000 archivos.');
    const normalizePath=(entry)=>{
      const original=String(entry.unsafeOriginalName??entry.name??'');
      if(!original||original.length>1024)throw new Error('El ZIP contiene una ruta vacía o demasiado larga.');
      if(original.includes(String.fromCharCode(92)))throw new Error('El ZIP contiene rutas con separadores no válidos.');
      if(/[\u0000-\u001f\u007f]/.test(original))throw new Error('El ZIP contiene caracteres de control en una ruta.');
      let path=original.replace(/^(?:\.\/)+/,'');
      if(entry.dir&&path.endsWith('/'))path=path.slice(0,-1);
      if(!path||path.startsWith('/')||/^[a-z]:/i.test(path)||path.startsWith('//'))throw new Error('El ZIP contiene una ruta absoluta o inválida.');
      const parts=path.split('/');
      if(parts.length>64||parts.some(part=>!part||part==='.'||part==='..'||part.length>255))throw new Error('El ZIP contiene una ruta insegura o demasiado profunda.');
      const parsedName=String(entry.name||'').replace(/^(?:\.\/)+/,'').replace(/\/$/,'').normalize('NFC');
      if(parsedName&&parsedName!==path.normalize('NFC'))throw new Error('El ZIP contiene una ruta alterada o ambigua.');
      return path.normalize('NFC');
    };
    const isNoise=(path)=>{
      const parts=path.split('/').map(part=>part.toLowerCase());
      const base=parts[parts.length-1]||'';
      return parts.includes('__macosx')||parts.includes('.spotlight-v100')||parts.includes('.trashes')||base==='.ds_store'||base==='thumbs.db'||base==='desktop.ini';
    };
    const records=[];
    for(const entry of allEntries){
      const path=normalizePath(entry);
      if(entry.dir)continue;
      if(isNoise(path))continue;
      const size=Number(entry._data?.uncompressedSize??entry._data?.length??0);
      if(!Number.isSafeInteger(size)||size<0)throw new Error('El ZIP declara un tamaño de archivo inválido.');
      if(size>LIMITS.entry)throw new Error('El archivo '+path+' supera 100 MB descomprimidos.');
      const rawMode=entry.unixPermissions;
      const mode=typeof rawMode==='string'?parseInt(rawMode,8):Number(rawMode||0);
      if(Number.isFinite(mode)&&(mode&0o170000)===0o120000)throw new Error('El ZIP contiene un enlace simbólico y no puede procesarse.');
      records.push({entry,path,size});

    }
    const totalExpanded=records.reduce((sum,item)=>sum+item.size,0);
    if(!Number.isSafeInteger(totalExpanded))throw new Error('El tamaño total descomprimido no es válido.');

      originalNames.add(key);
    }
    for(const item of records){
      const parts=item.path.split('/');
      for(let i=1;i<parts.length;i++){
        const parent=parts.slice(0,i).join('/').toLowerCase();
        if(originalNames.has(parent))throw new Error('El ZIP mezcla un archivo y una carpeta con el mismo nombre.');
      }
    }

    for(const item of normalizedRecords){

        phase:'validate',
        percent:((recordIndex+1)/Math.max(1,normalizedRecords.length))*100,

      compiled:false,compiledFiles:[],protected:false,rootPrefix,sourceFile:file,expandedBytes:normalizedExpanded,runtimeTarget:'MTA:SA',runtimeEvidence,mtaApiCount,fiveMApiCount
    };
    const configCount=files.filter(item=>item.kind==='config'||item.kind==='shared_lua').length;
    const shaderCount=files.filter(item=>item.kind==='shader').length;
    const scriptCount=files.filter(item=>item.kind.endsWith('_lua')||item.kind.endsWith('_luac')||item.kind==='lua_other').length;
    const bytes=n=>{
      n=Number(n)||0;
      if(n<1024)return n+' B';
      if(n<1048576)return (n/1024).toFixed(1)+' KB';
      return (n/1048576).toFixed(2)+' MB';
    };
    const rows=files.map(item=>{
      const label=item.kind==='server_lua'?'Lua servidor':item.kind==='client_lua'?'Lua cliente':item.kind==='shared_lua'?'Lua shared':item.kind==='config'?'Configuración':item.kind==='lua_other'?'Lua adicional':item.kind==='shader'?'Shader':'Archivo';
      return '<div class="zip-file dx-zip-entry dx-zip-static"><span class="dx-zip-name">'+esc(item.path)+'</span><small class="dx-zip-kind">'+label+' · '+bytes(item.bytes)+'</small></div>';
    }).join('');
    const analysisNode=ensureZipUiNode('zip-analysis','div');
    const fileListNode=ensureZipUiNode('zip-file-list','div');
    const protectButton=ensureZipUiNode('zip-protect','button');
    const compileButton=ensureZipUiNode('zip-compile','button');
    const downloadButton=ensureZipUiNode('zip-download','button');
    const statusNode=ensureZipUiNode('zip-status','p');
    for(const button of [protectButton,compileButton,downloadButton])button.type='button';
    analysisNode.innerHTML='<b>ZIP analizado · MTA:SA</b> · '+files.length+' archivos · '+scriptCount+' Lua · '+server.length+' server · '+client.length+' client · '+shaderCount+' shaders · '+configCount+' config/shared · '+bytes(normalizedExpanded);
    fileListNode.innerHTML='<div class="dx-zip-list-head"><strong>Archivos detectados</strong><span>'+files.length+'</span></div>'+rows;
    protectButton.disabled=false;
    protectButton.textContent='Proteger';
    compileButton.disabled=true;
    compileButton.textContent='Compilar';
    downloadButton.disabled=true;
    downloadButton.textContent='Preparar ZIP';
    statusNode.textContent='ZIP analizado.';
    const result={file,zip,entries,files,expanded:normalizedExpanded,metaPath:'meta.xml',rootPrefix,server:serverEntries,client:clientEntries,luaCount:lua.length,runtimeTarget:'MTA:SA',runtimeEvidence,mtaApiCount,fiveMApiCount};
    window.DRIPX_ZIP_SCAN_RESULT=result;
    window.DRIPX_ZIP_ROOT_PREFIX=rootPrefix;
    return result;
  }

  window.DRIPX_INJECT_LICENSE_GUARD = async function(){
    const build=state.zipBuild;
    if(!build?.zip||!build.protected||!build.compiled)throw new Error('El resource todavía no está protegido y compilado.');
    const guard=build.zip.file('dripx_server_license.lua');
    const config=build.zip.file('DRIPX-CFG.LUA');
    if(!guard||!config)throw new Error('Falta el guardián server o su configuración editable.');
    const source=await guard.async('string');
    const configSource=await config.async('string');
    if(source.split('DRIPX_LICENSE_KEY_PLACEHOLDER').length!==2||configSource.split('DRIPX_LICENSE_KEY_PLACEHOLDER').length!==2)throw new Error('El guardián o la configuración no tienen un marcador de licencia válido.');
    const retainedShared=(build.shared||[]).every(entry=>!!build.zip.file(entry.name)&&(/\.lua$/i.test(entry.name)||/\.luac$/i.test(entry.name)));
    if(!retainedShared)throw new Error('No se conservaron todos los archivos shared originales.');
    if(!build.productId)throw new Error('Falta el ID del producto.');
    build.licenseInjected=true;
    return {server_license_template:true,shared_files_preserved:true,compiled_shared_count:0,product_id:build.productId};
  };

  // Independent, non-destructive inventory. This does not invent or inject meta.xml
  // and does not require MTA scripts, shaders, or compiler-compatible Lua.
  window.DRIPX_INSPECT_ZIP = async function inspectOriginalZip(file) {
    const LIMITS = { compressed: 50_000_000, files: 3000, entries: 6000, entry: 100 * 1024 * 1024, expanded: 200 * 1024 * 1024, meta: 2 * 1024 * 1024 };
    if (!(file instanceof File) || !/\.zip$/i.test(String(file.name || "")) || !file.size) {
      throw new Error("Selecciona un archivo .zip con contenido.");
    }
    if (file.size > LIMITS.compressed) throw new Error("El ZIP supera el límite de 50 MB de almacenamiento.");
    if (!window.zip?.ZipReader) throw new Error("No se cargó el nuevo lector ZIP. Recarga Productos para actualizarlo.");
    const all = await dripxZipReadDirectory(file);
    const archive = { files: Object.fromEntries(all.map(entry=>[entry.name,entry])) };
    if (all.length > LIMITS.entries) throw new Error("El ZIP contiene demasiadas entradas.");
    const entries = all.filter(entry => !entry.dir);
    if (entries.length > LIMITS.files) throw new Error("El ZIP supera el límite de 3.000 archivos.");
    const seen = new Set();
    let expanded = 0;
    const files = [];
    let metaEntry = null;
    const shaderExtensions = new Set(["fx", "hlsl", "glsl"]);
    const configExtensions = new Set(["cfg", "ini", "conf", "json", "xml", "yaml", "yml", "toml", "properties"]);
    for (const entry of entries) {
      const info = entry;
      const original = String(info.unsafeOriginalName ?? entry.name ?? "");
      let path = original.replace(/^(?:\.\/)+/, "");
      if (!path || path.length > 1024 || path.startsWith("/") || path.includes("\\") ||
          /^[A-Za-z]:/.test(path) || /[\u0000-\u001f\u007f]/.test(path) ||
          path.split("/").some(part => !part || part === "." || part === ".." || part.length > 255)) {
        throw new Error("El ZIP contiene una ruta insegura o no compatible: " + original);
      }
      path = path.normalize("NFC");
      if (original !== entry.name && original.replace(/^(?:\.\/)+/, "") !== entry.name) {
        throw new Error("El ZIP contiene nombres ambiguos o alterados: " + original);
      }
      const key = path.toLowerCase();
      if (seen.has(key)) throw new Error("El ZIP contiene archivos duplicados: " + path);
      seen.add(key);
      const modeRaw = info.unixPermissions;
      const mode = typeof modeRaw === "string" ? parseInt(modeRaw, 8) : Number(modeRaw || 0);
      if (Number.isFinite(mode) && (mode & 0o170000) === 0o120000) {
        throw new Error("El ZIP contiene un enlace simbólico no permitido: " + path);
      }
      const bytes = Number(entry._data?.uncompressedSize ?? entry._data?.length ?? 0);
      if (!Number.isSafeInteger(bytes) || bytes < 0 || bytes > LIMITS.entry) {
        throw new Error("El tamaño indicado para " + path + " no es válido.");
      }
      expanded += bytes;
      if (!Number.isSafeInteger(expanded) || expanded > LIMITS.expanded) {
        throw new Error("El ZIP supera 200 MB descomprimidos.");
      }
      const base = path.split("/").pop() || "";
      const ext = (base.includes(".") ? base.split(".").pop() : "").toLowerCase();
      let kind = "file";
      if (base.toLowerCase() === "meta.xml") {
        kind = "meta";
        if (!metaEntry) metaEntry = { entry, path, bytes };
      } else if (shaderExtensions.has(ext)) {
        kind = "shader";
      } else if (ext === "lua" || ext === "luac") {
        kind = ext === "luac" ? "compiled_lua" : "lua";
      } else if (configExtensions.has(ext) ||
          /^(config|settings|options|language|locale)(?:[._-].*)?\.lua$/i.test(base) ||
          /(^|\/)(config|configs|settings)(\/|$)/i.test(path)) {
        kind = "config";
      }
      files.push({ path, bytes, extension: ext, kind });
    }
    // Fast index-only preview. Full decompression and CRC validation run once
    // in DRIPX_LOAD_ZIP after the administrator presses Trabajar.
    const metaCandidates = files.filter(item => item.kind === "meta");
    const rootMeta = metaCandidates.find(item => item.path.toLowerCase() === "meta.xml") || null;
    const warnings = [];
    let metaState = "found";
    if (!metaCandidates.length) {
      metaState = "missing";
      warnings.push({
        code: "META_XML_MISSING",
        message: "No se encontró meta.xml. El ZIP se abre y puede publicarse/descargarse; MTA:SA no podrá arrancar el resource hasta que se añada."
      });
    } else if (!rootMeta) {
      metaState = "nested";
      warnings.push({
        code: "META_XML_NOT_ROOT",
        message: "Se encontró meta.xml dentro de una carpeta, no en la raíz. Se muestran todos los archivos y no se bloquea la carga."
      });
    } else if (metaCandidates.length > 1) {
      metaState = "multiple";
      warnings.push({
        code: "META_XML_MULTIPLE",
        message: "Se encontraron varios archivos meta.xml. El inventario se muestra completo; revisa cuál debe ser el manifiesto del resource."
      });
    }
    if (rootMeta) {
      if (rootMeta.bytes > LIMITS.meta) {
        metaState = "too_large";
        warnings.push({ code: "META_XML_TOO_LARGE", message: "meta.xml supera 2 MB; se conserva como archivo, pero no se analiza su contenido." });
      } else {
        try {
          const text = await rootMeta.entry.async("string");
          if (/<!DOCTYPE|<!ENTITY/i.test(text)) {
            metaState = "unsafe";
            warnings.push({ code: "META_XML_UNSAFE", message: "meta.xml contiene declaraciones XML externas. Se muestra el ZIP sin interpretar ese manifiesto." });
          } else {
            const parsed = new DOMParser().parseFromString(text, "application/xml");
            if (parsed.querySelector("parsererror") || parsed.documentElement?.tagName?.toLowerCase() !== "meta") {
              metaState = "invalid";
              warnings.push({ code: "META_XML_INVALID", message: "meta.xml no contiene XML MTA válido. El inventario y la carga del ZIP no se bloquean." });
            }
          }
        } catch {
          metaState = "invalid";
          warnings.push({ code: "META_XML_UNREADABLE", message: "No se pudo interpretar meta.xml. El archivo permanece en el inventario y el ZIP no se bloquea." });
        }
      }
    }
    const shaderCount = files.filter(item => item.kind === "shader").length;
    if (!shaderCount) {
      warnings.push({
        code: "SHADER_FILES_MISSING",
        message: "No se encontraron archivos shader (.fx, .hlsl o .glsl). La ausencia de shaders no impide abrir, guardar ni descargar el ZIP."
      });
    }
    const counts = {
      total: files.length,
      lua: files.filter(item => item.kind === "lua").length,
      compiledLua: files.filter(item => item.kind === "compiled_lua").length,
      config: files.filter(item => item.kind === "config").length,
      shaders: shaderCount,
      meta: metaCandidates.length
    };
    const manifest = {
      inspector: "dripx-zipjs-balanced-index-v3",
      source_filename: file.name,
      source_bytes: file.size,
      expanded_bytes: expanded,
      files: files.map(item => ({ path: item.path, bytes: item.bytes, kind: item.kind, extension: item.extension })),
      counts,
      meta: { state: metaState, path: rootMeta?.path || metaCandidates[0]?.path || null },
      warnings
    };
    const result = { file, archive, entries, files, expanded, counts, metaState, metaPath: rootMeta?.path || null, warnings, manifest };
    window.DRIPX_ZIP_INVENTORY = result;
    return result;
  };
