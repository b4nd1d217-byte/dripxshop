  async function loadZipForProtection(file){
    const LIMITS={meta:2*1024*1024};
    if(!file||!Number.isFinite(Number(file.size))||Number(file.size)<=0)throw new Error('Selecciona un archivo ZIP válido.');
    if(!/\.zip$/i.test(String(file.name||'')))throw new Error('El archivo debe tener extensión .zip.');
    if(!window.JSZip)throw new Error('No se pudo cargar el lector ZIP. Recarga la página.');
    const archive=await JSZip.loadAsync(file,{checkCRC32:true,createFolders:false});
    const allEntries=Object.values(archive.files||{});
    if(!allEntries.length)throw new Error('El ZIP no contiene entradas.');
    const normalizePath=(entry)=>{
      const original=String(entry.unsafeOriginalName??entry.name??'');
      if(!original||original.length>1024)throw new Error('El ZIP contiene una ruta vacía o demasiado larga.');
      if(original.includes(String.fromCharCode(92)))throw new Error('El ZIP contiene rutas con separadores no válidos.');
      if(/[\u0000-\u001f\u007f]/.test(original))throw new Error('El ZIP contiene caracteres de control en una ruta.');
      let path=original.replace(/^(?:\.\/)+/,'');
      if(entry.dir&&path.endsWith('/'))path=path.slice(0,-1);
      if(!path||path.startsWith('/')||/^[a-z]:/i.test(path)||path.startsWith('//'))throw new Error('El ZIP contiene una ruta absoluta o inválida.');
      const parts=path.split('/');
      if(parts.length>64||parts.some(part=>!part||part==='.'||part==='..'||part.length>255))throw new Error('El ZIP contiene una ruta insegura o demasiado profunda.');
      const parsedName=String(entry.name||'').replace(/^(?:\.\/)+/,'').replace(/\/$/,'').normalize('NFC');
      if(parsedName&&parsedName!==path.normalize('NFC'))throw new Error('El ZIP contiene una ruta alterada o ambigua.');
      return path.normalize('NFC');
    };
    const isNoise=(path)=>{
      const parts=path.split('/').map(part=>part.toLowerCase());
      const base=parts[parts.length-1]||'';
      return parts.includes('__macosx')||parts.includes('.spotlight-v100')||parts.includes('.trashes')||base==='.ds_store'||base==='thumbs.db'||base==='desktop.ini';
    };
    const records=[];
    for(const entry of allEntries){
      const path=normalizePath(entry);
      if(entry.dir)continue;
      if(isNoise(path))continue;
      const size=Number(entry._data?.uncompressedSize??entry._data?.length??0);
      if(!Number.isSafeInteger(size)||size<0)throw new Error('El ZIP declara un tamaño de archivo inválido.');
      const rawMode=entry.unixPermissions;
      const mode=typeof rawMode==='string'?parseInt(rawMode,8):Number(rawMode||0);
      if(Number.isFinite(mode)&&(mode&0o170000)===0o120000)throw new Error('El ZIP contiene un enlace simbólico y no puede procesarse.');
      records.push({entry,path,size});

    }
    if(!records.length)throw new Error('El ZIP no contiene archivos utilizables.');
    const totalExpanded=records.reduce((sum,item)=>sum+item.size,0);
    if(!Number.isSafeInteger(totalExpanded))throw new Error('El tamaño total descomprimido no es válido.');
    const originalNames=new Set();
    for(const item of records){
      const key=item.path.toLowerCase();
      if(originalNames.has(key))throw new Error('El ZIP contiene nombres duplicados o rutas que solo cambian por mayúsculas.');
      originalNames.add(key);
    }
    for(const item of records){
      const parts=item.path.split('/');
      for(let i=1;i<parts.length;i++){
        const parent=parts.slice(0,i).join('/').toLowerCase();
        if(originalNames.has(parent))throw new Error('El ZIP mezcla un archivo y una carpeta con el mismo nombre.');
      }
    }
    const metaCandidates=records.filter(item=>item.path.toLowerCase()==='meta.xml'||item.path.toLowerCase().endsWith('/meta.xml'));
    if(metaCandidates.length!==1)throw new Error(metaCandidates.length?'El ZIP contiene más de un meta.xml. Debe incluir un solo resource MTA:SA.':'No se encontró meta.xml dentro del ZIP.');
    const metaRecord=metaCandidates[0];
    if(metaRecord.size>LIMITS.meta)throw new Error('meta.xml supera el límite de 2 MB.');
    const slash=metaRecord.path.lastIndexOf('/');
    const rootPrefix=slash>=0?metaRecord.path.slice(0,slash+1):'';
    const normalizedRecords=records.map(item=>{
      const relative=rootPrefix&&item.path.startsWith(rootPrefix)?item.path.slice(rootPrefix.length):item.path;
      const path=item===metaRecord?'meta.xml':relative;
      if(!path)throw new Error('El ZIP contiene una entrada vacía después de normalizar la carpeta principal.');
      return {...item,normalizedPath:path};
    });
    const normalizedNames=new Set();
    for(const item of normalizedRecords){
      const key=item.normalizedPath.toLowerCase();
      if(normalizedNames.has(key))throw new Error('La normalización de carpetas produce archivos duplicados.');
      normalizedNames.add(key);
    }
    for(const item of normalizedRecords){
      const parts=item.normalizedPath.split('/');
      for(let i=1;i<parts.length;i++){
        if(normalizedNames.has(parts.slice(0,i).join('/').toLowerCase()))throw new Error('La estructura del ZIP contiene un archivo donde debería existir una carpeta.');
      }
    }
    const zip=new JSZip();
    for(const item of normalizedRecords){
      const data=await item.entry.async('uint8array');
      if(!(data instanceof Uint8Array)||data.byteLength!==item.size)throw new Error('No se pudo leer íntegramente el archivo '+item.path+'.');
      zip.file(item.normalizedPath,data,{
        date:item.entry.date,
        createFolders:false,
        unixPermissions:item.entry.unixPermissions,
        dosPermissions:item.entry.dosPermissions
      });
    }
    const entries=Object.values(zip.files).filter(entry=>!entry.dir);
    const meta=zip.file('meta.xml');
    if(!meta)throw new Error('No se pudo reconstruir meta.xml en la raíz del resource.');
    const metaText=await meta.async('string');
    if(/<!DOCTYPE/i.test(metaText))throw new Error('meta.xml contiene una declaración DOCTYPE no permitida.');
    const parsed=new DOMParser().parseFromString(metaText,'application/xml');
    if(parsed.querySelector('parsererror')||parsed.documentElement?.tagName?.toLowerCase()!=='meta')throw new Error('meta.xml no es un XML válido de MTA:SA.');
    const normalizeScriptPath=value=>{
      let path=String(value||'').split(String.fromCharCode(92)).join('/');
      path=path.replace(/^(?:\.\/)+/,'');
      if(!path||path.startsWith('/')||/^[a-z]:/i.test(path)||/[\u0000-\u001f\u007f]/.test(path))throw new Error('meta.xml contiene una ruta de script inválida.');
      const parts=path.split('/');
      if(parts.length>64||parts.some(part=>!part||part==='.'||part==='..'||part.length>255))throw new Error('meta.xml contiene una ruta de script insegura.');
      return path.normalize('NFC');
    };
    const scripts=[];
    for(const node of parsed.querySelectorAll('script')){
      const originalSrc=String(node.getAttribute('src')||'');
      const src=normalizeScriptPath(originalSrc);
      const type=String(node.getAttribute('type')||'server').trim().toLowerCase();
      if(!['server','client','shared'].includes(type))throw new Error('meta.xml contiene un tipo de script no reconocido: '+type+'.');
      if(scripts.some(script=>script.src.toLowerCase()===src.toLowerCase()))throw new Error('meta.xml declara el mismo script más de una vez: '+src);
      scripts.push({src,originalSrc,type});
    }
    if(!scripts.length)throw new Error('meta.xml no declara scripts Lua.');
    const available=new Set(entries.map(entry=>entry.name));
    const missing=scripts.filter(script=>!available.has(script.src));
    if(missing.length)throw new Error('meta.xml referencia archivos que no existen en el ZIP: '+missing.slice(0,8).map(item=>item.src).join(', '));
    const scriptMap=new Map(scripts.map(script=>[script.src,script.type]));
    const scriptSrcByFile=new Map(scripts.map(script=>[script.src,script.originalSrc]));
    const reservedNames=new Set(['dripx_license.lua','dripx_guard_shared.lua','dripx_license.luac','dripx_license_guard.lua','dripx_manifest.json']);
    const reserved=entries.find(entry=>reservedNames.has(entry.name.split('/').pop().toLowerCase()));
    if(reserved)throw new Error('El ZIP contiene un archivo reservado por DRIPX: '+reserved.name+'. Renómbralo y vuelve a subirlo.');
    const configExtensions=new Set(['xml','json','ini','cfg','conf','yaml','yml','toml']);
    const files=entries.map(entry=>{
      const path=entry.name,lower=path.toLowerCase(),parts=path.split('/'),base=parts.pop()||'';
      const ext=base.includes('.')?base.slice(base.lastIndexOf('.')+1).toLowerCase():'';
      const scriptType=scriptMap.get(path)||'';
      const isLua=ext==='lua';
      const configFolder=parts.some(part=>/^(config|configs|settings|shared|language|languages|locale|locales)$/i.test(part));
      const configName=/^(config|settings|shared|language|locale)(?:[._-].*)?\.(lua|xml|json|ini|cfg|conf|ya?ml|toml)$/i.test(base);
      const isConfig=configExtensions.has(ext)||configFolder||configName;
      const isShader=['fx','hlsl','glsl'].includes(ext);
      const kind=isConfig?'config':(isLua&&scriptType==='shared'?'shared_lua':(isLua&&scriptType==='server'?'server_lua':(isLua&&scriptType==='client'?'client_lua':(isLua?'lua_other':(isShader?'shader':'file')))));
      return {
        path,bytes:Number(entry._data?.uncompressedSize??entry._data?.length??0),
        kind,scriptType,declared:!!scriptType,extension:ext
      };
    });
    const lua=entries.filter(entry=>entry.name.toLowerCase().endsWith('.lua'));
    const server=files.filter(item=>item.kind==='server_lua');
    const client=files.filter(item=>item.kind==='client_lua');
    const compileTargets=lua.filter(entry=>{
      const item=files.find(record=>record.path===entry.name);
      return item&&!['config','shared_lua'].includes(item.kind);
    });
    if(!lua.length||!compileTargets.length)throw new Error('El resource no contiene scripts Lua que puedan protegerse.');
    const serverEntries=entries.filter(entry=>server.some(item=>item.path===entry.name));
    const clientEntries=entries.filter(entry=>client.some(item=>item.path===entry.name));
    const normalizedExpanded=files.reduce((sum,item)=>sum+item.bytes,0);
    state.zipBuild={
      zip,metaName:'meta.xml',metaText,entries,lua,server:serverEntries,client:clientEntries,
      compileTargets,scriptSrcByFile,shared:entries.filter(entry=>scriptMap.get(entry.name)==='shared'),productId:crypto.randomUUID(),selected:new Set(compileTargets.map(entry=>entry.name)),
      compiled:false,protected:false,rootPrefix,sourceFile:file,expandedBytes:normalizedExpanded
    };
    const configCount=files.filter(item=>item.kind==='config'||item.kind==='shared_lua').length;
    const shaderCount=files.filter(item=>item.kind==='shader').length;
    const scriptCount=files.filter(item=>item.kind.endsWith('_lua')||item.kind==='lua_other').length;
    const bytes=n=>{
      n=Number(n)||0;
      if(n<1024)return n+' B';
      if(n<1048576)return (n/1024).toFixed(1)+' KB';
      return (n/1048576).toFixed(2)+' MB';
    };
    const rows=files.map(item=>{
      const label=item.kind==='server_lua'?'Lua servidor':item.kind==='client_lua'?'Lua cliente':item.kind==='shared_lua'?'Lua shared':item.kind==='config'?'Configuración':item.kind==='lua_other'?'Lua adicional':item.kind==='shader'?'Shader':'Archivo';
      return '<div class="zip-file dx-zip-entry dx-zip-static"><span class="dx-zip-name">'+esc(item.path)+'</span><small class="dx-zip-kind">'+label+' · '+bytes(item.bytes)+'</small></div>';
    }).join('');
    $('#zip-analysis').innerHTML='<b>ZIP analizado</b> · '+files.length+' archivos · '+scriptCount+' Lua · '+server.length+' server · '+client.length+' client · '+shaderCount+' shaders · '+configCount+' config/shared · '+bytes(normalizedExpanded);
    $('#zip-file-list').innerHTML='<div class="dx-zip-list-head"><strong>Archivos detectados</strong><span>'+files.length+'</span></div>'+rows;
    $('#zip-protect').disabled=false;
    $('#zip-protect').textContent='Proteger';
    $('#zip-compile').disabled=true;
    $('#zip-compile').textContent='Compilar';
    $('#zip-download').disabled=true;
    $('#zip-download').textContent='Preparar ZIP';
    $('#zip-status').textContent='ZIP analizado.';
    const result={file,zip,entries,files,expanded:normalizedExpanded,metaPath:'meta.xml',rootPrefix,server:serverEntries,client:clientEntries,luaCount:lua.length};
    window.DRIPX_ZIP_SCAN_RESULT=result;
    window.DRIPX_ZIP_ROOT_PREFIX=rootPrefix;
    return result;
  }
  window.DRIPX_LOAD_ZIP=loadZipForProtection;

  window.DRIPX_INJECT_LICENSE_GUARD = async function(){
    const build=state.zipBuild;
    if(!build?.zip||!build.protected||!build.compiled)throw new Error('El resource todavía no está protegido y compilado.');
    const shared=build.shared?.[0];
    if(!shared)throw new Error('El resource necesita un archivo Lua existente type="shared".');
    const source=await build.zip.file(shared.name).async('string');
    if(!source.includes('DRIPX_LICENSE_KEY_PLACEHOLDER'))throw new Error('No se encontró la marca de licencia en shared.');
    if(!build.productId)throw new Error('Falta el ID del producto.');
    build.licenseInjected=true;
    return {shared_license:true,product_id:build.productId};
  };
