  async function loadZipForProtection(file){
    if(!file)throw new Error('Selecciona un archivo ZIP.');
    if(!window.JSZip)throw new Error('No se pudo cargar el lector ZIP. Recarga la página e inténtalo de nuevo.');
    if(file.size>100*1024*1024)throw new Error('El ZIP supera el límite de 100 MB.');
    let zip=await JSZip.loadAsync(file,{checkCRC32:true});
    let entries=Object.values(zip.files).filter(x=>!x.dir&&!x.name.startsWith('__MACOSX/')&&!x.name.endsWith('/.DS_Store'));
    if(!entries.length)throw new Error('El ZIP no contiene archivos.');
    if(entries.length>3000)throw new Error('El ZIP contiene demasiados archivos (máximo 3.000).');
    if(entries.some(x=>x.name.startsWith('/')||x.name.includes('\\')||x.name.split('/').some(part=>part==='..'||part==='.'||!part)))throw new Error('El ZIP contiene rutas inseguras o inválidas.');
    const expanded=entries.reduce((n,x)=>n+Number(x._data?.uncompressedSize||0),0);
    if(expanded>200*1024*1024)throw new Error('El contenido descomprimido supera el límite de seguridad de 200 MB.');
    let meta=entries.find(x=>x.name.toLowerCase()==='meta.xml');
    let rootPrefix='';
    if(!meta){
      const nested=entries.filter(x=>x.name.toLowerCase().endsWith('/meta.xml'));
      if(nested.length===1){
        const prefix=nested[0].name.slice(0,nested[0].name.lastIndexOf('/')+1);
        if(!entries.every(x=>x.name.startsWith(prefix)))throw new Error('El ZIP mezcla archivos de varios recursos. Coloca un único resource MTA:SA en la raíz.');
        const normalized=new JSZip();
        for(const entry of entries){const relative=entry.name.slice(prefix.length);if(relative)normalized.file(relative,entry.async('uint8array'),{date:entry.date,createFolders:false});}
        zip=normalized;rootPrefix=prefix;entries=Object.values(zip.files).filter(x=>!x.dir);meta=entries.find(x=>x.name.toLowerCase()==='meta.xml');
      }
    }
    if(!meta)throw new Error('No se encontró meta.xml. El ZIP debe contener un resource MTA:SA con meta.xml en la raíz o dentro de una única carpeta principal.');
    const metaText=await meta.async('string');
    const parsed=new DOMParser().parseFromString(metaText,'application/xml');
    if(parsed.querySelector('parsererror')||parsed.documentElement?.tagName.toLowerCase()!=='meta')throw new Error('meta.xml no es un XML válido de MTA:SA.');
    const normalizePath=v=>{let s=String(v||'').split(String.fromCharCode(92)).join('/');while(s.startsWith('./'))s=s.slice(2);return s;};
    const scripts=[...parsed.querySelectorAll('script')].map(node=>({src:normalizePath(node.getAttribute('src')),originalSrc:String(node.getAttribute('src')||''),type:(node.getAttribute('type')||'server').toLowerCase()}));
    if(!scripts.length)throw new Error('meta.xml no declara scripts Lua.');
    if(scripts.some(x=>!x.src||x.src.startsWith('/')||x.src.split('/').some(part=>part==='..'||part==='.'||!part)))throw new Error('meta.xml contiene rutas de scripts no válidas.');
    const entryNames=new Set(entries.map(x=>x.name));
    const missing=scripts.filter(x=>!entryNames.has(x.src));
    if(missing.length)throw new Error('meta.xml referencia scripts que no existen en el ZIP: '+missing.map(x=>x.src).slice(0,8).join(', '));
    const scriptByPath=new Map(scripts.map(x=>[x.src,x]));
    const scriptSrcByFile=new Map(scripts.map(x=>[x.src,x.originalSrc]));
    const lua=entries.filter(x=>x.name.toLowerCase().endsWith('.lua'));
    const isConfigLua=x=>/(^|\/)(config|settings|shared|language|locale)[^/]*\.lua$/i.test(x.name);
    const server=lua.filter(x=>{const script=scriptByPath.get(x.name);return script&&script.type==='server'&&!isConfigLua(x);});
    const client=lua.filter(x=>{const script=scriptByPath.get(x.name);return script&&script.type==='client'&&!isConfigLua(x);});
    const compileTargets=[...server,...client];
    if(!server.length)throw new Error('El resource necesita al menos un Lua de servidor para validar la licencia DRIPX.');
    if(!compileTargets.length)throw new Error('No se encontraron Lua server/client declarados en meta.xml.');
    state.zipBuild={zip,metaName:meta.name,metaText,entries,lua,server,client,compileTargets,scriptSrcByFile,selected:new Set(compileTargets.map(x=>x.name)),compiled:false,protected:false,rootPrefix};
    const bytes=n=>{n=Number(n)||0;if(n<1024)return n+' B';if(n<1048576)return (n/1024).toFixed(1)+' KB';return (n/1048576).toFixed(2)+' MB';};
    const rows=entries.map(entry=>{
      const name=entry.name,lower=name.toLowerCase(),script=scriptByPath.get(name),ext=lower.split('.').pop();
      const isServer=server.some(x=>x.name===name),isClient=client.some(x=>x.name===name),isLua=lower.endsWith('.lua'),isShader=['fx','hlsl','glsl'].includes(ext),isConfig=/(^|\/)(config|settings|shared|language|locale)[^/]*\.(lua|xml|json|ini|cfg|conf|ya?ml)$/i.test(name)||['xml','json','ini','cfg','conf','yaml','yml'].includes(ext);
      const kind=isServer?'Lua servidor · compilable':(isClient?'Lua client · compilable':(isConfig?'Configuración':(isLua?'Lua shared/no declarado':(isShader?'Shader':'Archivo'))));
      const size=bytes(entry._data?.uncompressedSize||0);
      if(isServer||isClient)return '<label class="zip-file dx-zip-entry"><input type="checkbox" data-zip-lua="'+esc(name)+'" checked><span class="dx-zip-name">'+esc(name)+'</span><small class="dx-zip-kind">'+kind+' · '+size+'</small></label>';
      return '<div class="zip-file dx-zip-entry dx-zip-static"><span class="dx-zip-name">'+esc(name)+'</span><small class="dx-zip-kind">'+kind+' · '+size+(script?' · '+esc(script.type):'')+'</small></div>';
    }).join('');
    const configCount=entries.filter(x=>/(^|\/)(config|settings|shared|language|locale)[^/]*\.(lua|xml|json|ini|cfg|conf|ya?ml)$/i.test(x.name)||['xml','json','ini','cfg','conf','yaml','yml'].includes(x.name.toLowerCase().split('.').pop())).length;
    const shaderCount=entries.filter(x=>['fx','hlsl','glsl'].includes(x.name.toLowerCase().split('.').pop())).length;
    const clientCount=client.length;
    $('#zip-analysis').innerHTML='<b>ZIP leído correctamente</b><br>'+entries.length+' archivos · '+lua.length+' Lua · '+server.length+' Lua servidor · '+clientCount+' Lua client/shared · '+shaderCount+' shaders · '+configCount+' archivos de configuración'+(rootPrefix?'<br>Carpeta principal normalizada: '+esc(rootPrefix):'');
    $('#zip-file-list').innerHTML='<div class="dx-zip-list-head"><strong>Inventario completo del resource</strong><span>'+entries.length+' archivos</span></div>'+rows;
    $('#zip-protect').disabled=false;$('#zip-protect').textContent='Proteger Lua';$('#zip-compile').disabled=true;$('#zip-download').disabled=true;
    $('#zip-status').textContent='Selecciona los Lua de servidor y client que se compilarán/protegerán. Shared/config, shaders, XML e imágenes se mantienen sin cambiar.';
    window.DRIPX_ZIP_ROOT_PREFIX=rootPrefix;
  }
  window.DRIPX_LOAD_ZIP=loadZipForProtection;
